# Demonstrates: while loop, wait, retry pattern
name: Service Health Check with Retry
vars:
  service_url: "http://localhost:8080/health"
  max_retries: 5
steps:
  - log: "Starting health check for ${Host_IP}"
  - set: { expression: retry = 0 }
  - while:
      condition: retry < max_retries and healthy is empty
      do:
        - send:
            command: curl -s -w "%{http_code}" ${service_url}
            capture: http_code
